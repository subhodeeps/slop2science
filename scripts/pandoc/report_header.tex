% Included by scripts/report_pdf.sh (pandoc -H). Fixes at the level of the pipeline, so the
% Markdown stays readable in a KaTeX previewer. Needs xelatex or lualatex (fontspec).
%
% Fenced code blocks: pandoc writes plain `verbatim`, which never wraps. A long command then
% overflows the margin. fvextra's Verbatim with breaklines wraps it with a continuation mark.
\usepackage{fvextra}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{breaklines,breakanywhere}
% Tagged code blocks use pandoc's `Highlighting` environment. Wrap those too.
\makeatletter
\@ifundefined{Highlighting}{}{%
  \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},breaklines,breakanywhere}}
\makeatother
% A glyph that the chosen fonts lack stops the build with an error ("Missing character"). Value
% 2 only writes a log line, and the symbol vanishes from the PDF in silence. Value 3 needs
% TeX Live 2021 or newer.
\tracinglostchars=3
% fit_display_math.lua wraps each long display equation in adjustbox (max width).
\usepackage{adjustbox}
% Inline code can contain Greek letters and box-drawing characters. Latin Modern Mono has
% neither. DejaVu Sans Mono has both, so use it when the system has it.
\IfFontExistsTF{DejaVu Sans Mono}{\setmonofont{DejaVu Sans Mono}[Scale=0.85]}{}
\emergencystretch=3em
% Tables: one size smaller than the body. break_table_code.lua sets the column widths.
\AtBeginEnvironment{longtable}{\small}
% Greek letters typed directly in prose are not in Latin Modern Roman. Map each one to its
% math-italic glyph, so that it renders and does not vanish.
\usepackage{newunicodechar}
\newunicodechar{α}{\ensuremath{\alpha}}\newunicodechar{β}{\ensuremath{\beta}}
\newunicodechar{γ}{\ensuremath{\gamma}}\newunicodechar{δ}{\ensuremath{\delta}}
\newunicodechar{ε}{\ensuremath{\varepsilon}}\newunicodechar{κ}{\ensuremath{\kappa}}
\newunicodechar{λ}{\ensuremath{\lambda}}\newunicodechar{μ}{\ensuremath{\mu}}
\newunicodechar{ν}{\ensuremath{\nu}}\newunicodechar{ρ}{\ensuremath{\rho}}
\newunicodechar{σ}{\ensuremath{\sigma}}\newunicodechar{ψ}{\ensuremath{\psi}}
\newunicodechar{ω}{\ensuremath{\omega}}\newunicodechar{Γ}{\ensuremath{\Gamma}}
\newunicodechar{Δ}{\ensuremath{\Delta}}\newunicodechar{ℓ}{\ensuremath{\ell}}
